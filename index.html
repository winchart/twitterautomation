const LOCAL_AGENT_URL = "http://127.0.0.1:5055/start-posting"; const MASTER_KEY = "MY_TWITTER_SECRET_MASTER_KEY_2026"; async function triggerAutomation() { const formData = new FormData(); formData.append("post_text", "Hello from my website!"); formData.append("profiles", JSON.stringify(["user1", "user2"])); formData.append("master_key", MASTER_KEY); // চাবি পাস করা হলো try { let response = await fetch(LOCAL_AGENT_URL, { method: "POST", headers: { "X-Master-Key": MASTER_KEY // অথবা হেডার দিয়ে অথেন্টিকেশন }, body: formData }); let result = await response.json(); alert(result.message); } catch (error) { console.error("Local agent not running or connection failed:", error); alert("লোকাল এজেন্ট পিসিতে চালু করা নেই!"); } }
